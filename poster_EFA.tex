\Message{Does the leveraged positioning of hedge funds amplify sovereign bond yield sensitivity to shocks, and under what conditions?}
\vspace{2mm}
{\fHead\color{rust}\raisebox{1pt}{$\blacktriangleright$}\hspace{0.5em}\textbf{Yes. Positioning accounts for more than a quarter of the shock-day yield move, the overshooting is temporary, and the directionality of the book governs its strength.}\par}

% ============================ MECHANISM ============================
\begin{Panel}
\Section{How leveraged positioning amplifies yields}

{\fBody Leveraged intermediaries manage their balance sheet against the risk environment (Adrian and Shin 2014). The same rule produces opposite position adjustments depending on the sign of the shock.\par}
\vspace{6mm}

\begin{center}
\begin{tikzpicture}[box/.style={draw=navy,line width=1.3pt,rounded corners=3mm,align=center,
    inner xsep=9mm,inner ysep=5mm,text=ink,font=\fBody}]
  \node[box,draw=rust,fill=paler] (con)
    {{\fBody\bfseries\color{rust}Constraining}\\[1.5mm]
     fund \textbf{reduces} exposure};
  \node[box,fill=pale,text=navy,font=\fBody\bfseries,right=16mm of con] (top)
    {Shock meets a predetermined\\[1mm] leveraged position};
  \node[box,draw=grn,fill=paleg,right=16mm of top] (rel)
    {{\fBody\bfseries\color{grn}Relaxing}\\[1.5mm]
     fund \textbf{expands} exposure};
  \draw[-{Latex[length=5mm]},line width=1.5pt,navy] (top.west) -- (con.east);
  \draw[-{Latex[length=5mm]},line width=1.5pt,navy] (top.east) -- (rel.west);
\end{tikzpicture}
\end{center}

\vspace{3mm}
{\fBody\textbf{\color{navy}Numerical example} (margin 2\%, duration 7y, sector holds 3\% of the issue):\quad 6 bp surprise $\rightarrow$ price $-0.42\%$ $\rightarrow$ capital buffer $\color{rust}\mathbf{-21\%}$ $\rightarrow$ trading pressure $0.63\%$ of the issue $\rightarrow$ $\color{rust}\mathbf{+3.15}$ \textbf{\color{rust}bp} extra yield move.\par}

\Message{In both states the resulting trading pushes yields further in the direction of the initial shock.}
\end{Panel}

% ============================ FINDING 4 ============================
\begin{Panel}
\Section{Finding 4: The directionality of the book governs the strength}

\begin{minipage}[c]{272mm}
  \centering
  \includegraphics[width=264mm]{Figures/motivation_chart.png}
  \FigNote{Aggregate net repo positions in German (DE) and Italian (IT) sovereigns. Around the hiking cycle the sector runs a hedged book that is long IT and short DE. During the hiking cycle it runs a one-sided directional book.}
\end{minipage}\hfill
\begin{minipage}[c]{300mm}
  {\fBody A fund manages one pool of capital against its whole book. A duration-hedged book barely loses capital on a level move while a directional book absorbs the full hit.\par}
  \vspace{6mm}
  \fTab\centering
  \begin{tabular}{l c}
    \toprule
    $\Delta$ yield (bp), relative to no HF & Coefficient\\
    \midrule

    \midrule
    Q1 most hedged $\times$ Shock & 0.057\\
    Q2 $\times$ Shock & 0.137$^{***}$\\
    Q3 $\times$ Shock & 0.361$^{***}$\\
    \textbf{Q4 most directional $\times$ Shock} & \textbf{0.367}$^{***}$\\
    \bottomrule
  \end{tabular}
  \FigNote{\centering Bond and duration $\times$ country $\times$ date FE with controls. Wald tests reject equality of the hedged and directional groups ($F=24.8$) and of the extreme quartiles ($F=28.3$).\par}
\end{minipage}

\Message{The amplification rises monotonically with directionality and vanishes when the book is hedged.}
\end{Panel}

% ========================= CONCLUSION =========================
\begin{Panel}
\Section{Conclusion: What this means for risk monitoring}

\begin{minipage}[t]{300mm}\fBody
\begin{itemize}
  \item \textbf{More than a quarter} of the shock-day yield move reflects positioning and reverts within days.
  \item The amplification is \textbf{symmetric} across constraining and relaxing states.
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{300mm}\fBody
\begin{itemize}
  \item It is \textbf{not confined to stress episodes}.
  \item The \textbf{\color{rust}directionality} of the book is the key risk metric beyond position size.
\end{itemize}
\end{minipage}
\end{Panel}
